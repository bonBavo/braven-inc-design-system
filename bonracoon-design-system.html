Tailwind Theme

Drop straight into tailwind.config.js.

theme: {
  extend: {
    colors: {
      primary: "#f97316",
      secondary: "#1e293b",
      neutral: "#171717",
      success: "#22c55e",
      warning: "#facc15",
      danger: "#ef4444",
      info: "#38bdf8",
    },
    borderRadius: {
      xl: "1rem",
      "2xl": "1.5rem",
    },
    boxShadow: {
      glow: "0 0 30px rgb(249 115 22 / 0.4)",
    },
  },
}
